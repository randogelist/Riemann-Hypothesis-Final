function hashOutput(output, code, path) {
  if(code!==0) error("SHA256 command failed for " + path + " : " + output);
  var lines=output.split(/\r?\n/), found=[];
  for(var i=0;i<lines.length;i++) {var x=lines[i].replace(/\s/g,"");if(/^[a-fA-F0-9]{64}$/.test(x))found.push(x.toLowerCase());}
  if(found.length!==1)error("Unrecognized certutil SHA256 result for " + path);
  return found[0];
}
function fileHash(path) {
  if(!exists(path))error("Cannot fingerprint missing file: " + path);
  var proc=shell.Exec(quote(certutil)+" -hashfile "+quote(path)+" SHA256");
  var output=proc.StdOut.ReadAll()+proc.StdErr.ReadAll();
  while(proc.Status===0)WScript.Sleep(10);
  return hashOutput(output,proc.ExitCode,path);
}

function parallelHashes() {
  var todo=keys(hashFiles), active=[], cursor=0, done=0, lastReport=0;
  log("DEPENDENCY_CONTENT_FILES="+todo.length);
  log("DEPENDENCY_HASH_MODE=SHA256_CONTENT_PARALLEL_"+jobs);
  while(cursor<todo.length || active.length){
    while(cursor<todo.length && active.length<jobs){
      var key=todo[cursor++],p=hashFiles[key];
      active.push({key:key,path:p,proc:shell.Exec(quote(certutil)+" -hashfile "+quote(p)+" SHA256")});
    }
    for(var i=active.length-1;i>=0;i--){
      var x=active[i]; if(x.proc.Status===0)continue;
      var output=x.proc.StdOut.ReadAll()+x.proc.StdErr.ReadAll();
      fileHashes[x.key]=hashOutput(output,x.proc.ExitCode,x.path);
      active.splice(i,1);done++;
    }
    if(done-lastReport>=256 || done===todo.length){log("DEPENDENCY_HASH_PROGRESS="+done+"/"+todo.length);lastReport=done;}
    if(active.length)WScript.Sleep(10);
  }
}
function sourceClean(s, strict) {
  /* Preserve newlines while removing nested comments and string literals. */
  var out="",depth=0,inString=false,i=0;
  while(i<s.length){var a=s.charAt(i),b=s.charAt(i+1);
    if(depth){if(a==="/"&&b==="-"){depth++;out+="  ";i+=2;}else if(a==="-"&&b==="/"){depth--;out+="  ";i+=2;}else{out+=a==="\n"?"\n":" ";i++;}continue;}
    if(inString){if(a==="\\"){out+="  ";i+=2;}else{if(a==='"')inString=false;out+=a==="\n"?"\n":" ";i++;}continue;}
    if(a==="/"&&b==="-"){depth=1;out+="  ";i+=2;continue;}
    if(a==="-"&&b==="-"){while(i<s.length&&s.charAt(i)!=="\n"){out+=" ";i++;}continue;}
    if(a==='"'){inString=true;out+=" ";i++;continue;}
    out+=a;i++;
  }
  if(strict!==false && (depth||inString))error("Unclosed comment or string in Lean source scan.");
  return out;
}
function importsOf(source) {

}

  var re=/'([^']+)'\s+(?:depends on axioms:\s*\[([^\]]*)\]|does not depend on any axioms)/g;
  var m,found={},allowed={"propext":true,"Classical.choice":true,"Quot.sound":true};
  while((m=re.exec(contents))!==null){
    if(found[m[1]])error("Duplicate axiom record for "+m[1]);found[m[1]]=true;
    var axioms=typeof m[2]==="string"?trim(m[2]):"";
    if(axioms){var a=axioms.split(/\s*,\s*/);for(var i=0;i<a.length;i++)if(!allowed[trim(a[i])])error("Unapproved axiom "+a[i]+" in "+m[1]);}
  }
  for(var j=0;j<expected.length;j++)if(!found[expected[j]])error("Missing exact axiom-audit record for "+expected[j]);
  return expected.length;
}
function moduleInput(name) {
  var n=modules[name],s=SCHEMA+"\nTARGET="+TARGET+"\nTOOLCHAIN="+globalKey+"\nMODULE="+name+"\n";
  s+="SOURCE="+fileHashes[keyPath(n.source)]+"\nAUDIT="+auditTargets[name].join(",")+"\n";
  var deps=n.imports.slice().sort();
  for(var i=0;i<deps.length;i++){
    if(modules[deps[i]].local){if(!moduleKeys[deps[i]])error("Local dependency not verified: "+deps[i]);s+=deps[i]+"="+moduleKeys[deps[i]]+";"+artifactHashes[deps[i]]+"\n";}
    else s+=deps[i]+"="+externalKey(deps[i],{})+"\n";
  }
  return {key:textHash(s),manifest:s};
}

}
function restoreAnchor(name,input) {
  var parent=anchorRoot+"\\"+name+"\\"+input.key,entries=subfolders(parent);
  for(var i=entries.length-1;i>=0;i--){
    try{
    var folder=entries[i],receiptFile=folder+"\\receipt.txt";
      if(/\.pending$/.test(folder))continue;
      if(!exists(receiptFile))continue;
      var r=parseReceipt(read(receiptFile));
      if(r.SCHEMA!==SCHEMA||r.TARGET!==TARGET||r.MODULE!==name||r.INPUT_SHA256!==input.key||r.BUILD_PASS!=="1"||r.AUDIT_PASS!=="1"||r.LEAN_VERSION!=="4.33.0")continue;
      if(r.SOURCE_SHA256!==fileHashes[keyPath(modules[name].source)]||fileHash(folder+"\\source.lean")!==r.SOURCE_SHA256)continue;
      if(fileHash(folder+"\\audit.log")!==r.AUDIT_LOG_SHA256)continue;
      checkAxioms(read(folder+"\\audit.log"),auditTargets[name]);
      if(read(folder+"\\input.txt")!==input.manifest)continue;
      var artifacts=read(folder+"\\artifacts.tsv").split(/\r?\n/),verified=[],canonical="";
      for(var j=0;j<artifacts.length;j++){
        if(!trim(artifacts[j]))continue;var p=artifacts[j].split("|");
        if(p.length!==2||!/^[A-Za-z0-9_.]+$/.test(p[0])||!/^[a-f0-9]{64}$/.test(p[1]))error("Invalid anchor artifact manifest.");
        if(fileHash(folder+"\\artifacts\\"+p[0])!==p[1])error("Anchor artifact content hash mismatch.");
        verified.push(p[0]);canonical+=p[0]+"="+p[1]+"\n";
      }
      if(!verified.length||textHash(canonical)!==r.ARTIFACTS_SHA256)continue;
      var required=fso.GetFileName(name.replace(/\./g,"\\"))+".olean",hasMain=false;
      for(j=0;j<verified.length;j++)if(verified[j]===required)hasMain=true;
      if(!hasMain)continue;
      var output=clearOutputs(name),outputDir=fso.GetParentFolderName(output);mkdir(outputDir);
      for(j=0;j<verified.length;j++)fso.CopyFile(folder+"\\artifacts\\"+verified[j],outputDir+"\\"+verified[j],false);
      moduleKeys[name]=input.key;artifactHashes[name]=r.ARTIFACTS_SHA256;buildStates[name]="FROZEN_REUSED";
      append(auditLogPath,"FROZEN_AUDIT="+name+"\r\n"+read(folder+"\\audit.log")+"\r\n");
      log("FROZEN_REUSE="+name);log("ANCHOR_RECEIPT="+receiptFile);reused++;audited+=auditTargets[name].length;return true;
    }catch(e){log("ANCHOR_REJECTED="+entries[i]+" : "+e.message);}
  }
  return false;
}
function saveAnchor(name,input,auditLog,output) {
  var parent=anchorRoot+"\\"+name+"\\"+input.key;mkdir(parent);
  var dest=parent+"\\"+runTag,temp=dest+".pending";
  if(directory(dest)||directory(temp))error("Unique anchor directory collision.");mkdir(temp+"\\artifacts");
  var files=artifactFamily(output),manifest="",canonical="";
  if(!exists(output))error("Compiler did not create "+output);
  for(var i=0;i<files.length;i++){
    var fileName=fso.GetFileName(files[i]),hash=fileHash(files[i]);
    fso.CopyFile(files[i],temp+"\\artifacts\\"+fileName,false);
    if(fileHash(temp+"\\artifacts\\"+fileName)!==hash)error("Anchor output copy verification failed.");
    manifest+=fileName+"|"+hash+"\n";canonical+=fileName+"="+hash+"\n";
  }
  fso.CopyFile(modules[name].source,temp+"\\source.lean",false);
  fso.CopyFile(auditLog,temp+"\\audit.log",false);
  if(fileHash(temp+"\\source.lean")!==fileHashes[keyPath(modules[name].source)])error("Source changed during compilation; no anchor is accepted.");
  write(temp+"\\input.txt",input.manifest);write(temp+"\\artifacts.tsv",manifest);
  var artifactKey=textHash(canonical),auditHash=fileHash(temp+"\\audit.log");
  var receipt="SCHEMA="+SCHEMA+"\nTARGET="+TARGET+"\nLEAN_VERSION=4.33.0\nMODULE="+name+"\nINPUT_SHA256="+input.key+
    "\nSOURCE_SHA256="+fileHashes[keyPath(modules[name].source)]+"\nARTIFACTS_SHA256="+artifactKey+"\nAUDIT_LOG_SHA256="+auditHash+
    "\nBUILD_PASS=1\nAUDIT_PASS=1\nAUDIT_TARGET_COUNT="+auditTargets[name].length+"\nTIME_UTC="+stamp()+"\n";
  write(temp+"\\receipt.txt",receipt);fso.MoveFolder(temp,dest);
  moduleKeys[name]=input.key;artifactHashes[name]=artifactKey;buildStates[name]="COMPILED_AUDITED";
  log("ANCHOR_SAVED="+dest+"\\receipt.txt");
}
function processModule(name) {
  var n=modules[name],i;
  for(i=0;i<n.imports.length;i++)if(modules[n.imports[i]].local&&!moduleKeys[n.imports[i]]){
    clearOutputs(name);buildStates[name]="BLOCKED_DEPENDENCY";blocked++;log("MODULE_BLOCKED="+name+" DEPENDENCY="+n.imports[i]);return;
  }
  var input=moduleInput(name);if(restoreAnchor(name,input))return;
  var output=clearOutputs(name);mkdir(fso.GetParentFolderName(output));
  var compileLog=helpers+"\\"+name+".compile.log",auditSource=root+"\\RunnerAudit\\"+name.replace(/\./g,"_")+".lean",auditLog=helpers+"\\"+name+".audit.log";
  log("COMPILE_START="+name);
  if(runToFile(compiler,["-j"+jobs,"-o",output,n.source],compileLog)!==0){
    log(read(compileLog));clearOutputs(name);buildStates[name]="COMPILE_FAILED";failed++;log("COMPILE_FAIL="+name);return;
  }
  var compileContents=read(compileLog);if(compileContents)log(compileContents);
  if(/sorryAx/.test(compileContents))error("Compiler reported sorryAx in "+name);
  var audit="import "+name+"\n";
  for(i=0;i<auditTargets[name].length;i++)audit+="#check "+auditTargets[name][i]+"\n#print axioms "+auditTargets[name][i]+"\n";
  write(auditSource,audit);log("AXIOM_AUDIT_START="+name);
  if(runToFile(compiler,["-j"+jobs,auditSource],auditLog)!==0){
    log(read(auditLog));clearOutputs(name);buildStates[name]="AUDIT_COMPILE_FAILED";failed++;return;
  }
  try{audited+=checkAxioms(read(auditLog),auditTargets[name]);}
  catch(e){clearOutputs(name);buildStates[name]="AUDIT_FAILED";failed++;log("AXIOM_AUDIT_FAIL="+name+" : "+e.message);return;}
  append(auditLogPath,"MODULE="+name+"\r\n"+read(auditLog)+"\r\n");
  saveAnchor(name,input,auditLog,output);compiled++;log("COMPILE_AND_AXIOM_AUDIT_PASS="+name);
}
function finalAudit() {
  var input=moduleInput("Audit"),expected=auditTargets.Audit;
  if(restoreAnchor("Audit",input)){log("FINAL_AUDIT_DECLARATIONS="+expected.length);return;}
  var output=clearOutputs("Audit"),file=helpers+"\\final-audit.log";
  log("FINAL_AUDIT_START=Audit.lean");
  if(runToFile(compiler,["-j"+jobs,"-o",output,root+"\\Audit.lean"],file)!==0)error("Final Audit.lean failed: "+read(file));
  checkAxioms(read(file),expected);append(auditLogPath,"FINAL_AUDIT\r\n"+read(file)+"\r\n");
  saveAnchor("Audit",input,file,output);compiled++;audited+=expected.length;
  log("FINAL_AUDIT_DECLARATIONS="+expected.length);
}
